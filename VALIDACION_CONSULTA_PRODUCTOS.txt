function findMatchingProducts(query: string, availableProducts: any[]): any[] {
  if (!query || !availableProducts) return [];
  const queryLower = query.toLowerCase().trim();
  return availableProducts.filter(p =>
    p.name?.toLowerCase().includes(queryLower) ||
    p.description?.toLowerCase().includes(queryLower) ||
    p.category?.toLowerCase().includes(queryLower)
  );
}

• Busca productos que coincidan con lo que el cliente pregunta
• Busca en: nombre, descripción y categoría
